389	        """Keep lane order independent of the estimated strength ranking."""
390	        roles = {"TOP": 0, "JUNGLE": 1, "MIDDLE": 2, "BOTTOM": 3, "UTILITY": 4,
391	                 "MID": 2, "BOT": 3, "SUPPORT": 4}
392	        players = self._last_session.get("players", {})
393	        matchups = self._last_session.get("lane_matchups", {})
394	        enemy_roles = {matchup.get("enemy_key"): role for role, matchup in matchups.items()
395	                       if isinstance(matchup, dict) and matchup.get("enemy_key")}
396	
397	        def position(threat):
398	            key = threat.get("key")
399	            role = enemy_roles.get(key) or players.get(key, {}).get("role") or threat.get("role", "")
400	            return roles.get(str(role).upper(), len(roles))
401	
402	        return sorted(threats, key=position)
